import os

def simple_messaging_app():
    filename = "message.txt"

    try:
        with open(filename, 'x') as f:
            print(f"Success: '{filename}' created successfully.")
    except FileExistsError:
        print(f"Error: The file '{filename}' already exists.")
    except Exception as e:
        print(f"An unexpected error occurred: {e}")

    while True:
        print("\n--- Simple Messaging App ---")
        print("1. Send a message (append to file)")
        print("2. View all messages (read from file)")
        print("3. Exit the program")
        
        choice = input("Select an option (1-3): ")

        if choice == '1':
            msg = input("Enter your message: ")
            try:
                with open(filename, 'a') as f:
                    f.write(msg + "\n")
                print("Message sent!")
            except Exception as e:
                print(f"Error saving message: {e}")

        elif choice == '2':
            print("\n--- Message History ---")
            try:
                with open(filename, 'r') as f:
                    content = f.read()
                    if content:
                        print(content)
                    else:
                        print("(No messages found)")
            except FileNotFoundError:
                print("Error: The file does not exist yet.")
            except Exception as e:
                print(f"Error reading messages: {e}")

        elif choice == '3':
            print("Exiting... Goodbye!")
            break
        
        else:
            print("Invalid selection. Please try again.")

if __name__ == "__main__":
    simple_messaging_app()
